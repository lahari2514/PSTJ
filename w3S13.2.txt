import java.util.*;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt(); // number of lines
        List<List<Integer>> lists = new ArrayList<>();

        // Read n lines
        for (int i = 0; i < n; i++) {
            int d = sc.nextInt(); // number of integers in this line
            List<Integer> line = new ArrayList<>();
            for (int j = 0; j < d; j++) {
                line.add(sc.nextInt());
            }
            lists.add(line);
        }

        int q = sc.nextInt(); // number of queries
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt(); // line number (1-based)
            int y = sc.nextInt(); // position in line (1-based)

            // Check bounds safely
            if (x <= lists.size() && y <= lists.get(x - 1).size()) {
                System.out.println(lists.get(x - 1).get(y - 1));
            } else {
                System.out.println("ERROR!");
            }
        }
    }
}
