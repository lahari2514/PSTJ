class Result {

    public static List<Integer> freqQuery(List<List<Integer>> queries) {

        // Stores: number -> frequency
        HashMap<Integer, Integer> freq = new HashMap<>();

        // Stores: frequency -> number of elements having that frequency
        HashMap<Integer, Integer> count = new HashMap<>();

        List<Integer> result = new ArrayList<>();

        for (List<Integer> query : queries) {

            int operation = query.get(0);
            int value = query.get(1);

            if (operation == 1) {

                // Insert value
                int oldFreq = freq.getOrDefault(value, 0);
                int newFreq = oldFreq + 1;

                freq.put(value, newFreq);

                // Remove from old frequency
                if (oldFreq > 0) {
                    count.put(oldFreq, count.get(oldFreq) - 1);
                }

                // Add to new frequency
                count.put(newFreq,
                        count.getOrDefault(newFreq, 0) + 1);
