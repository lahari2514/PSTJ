import java.io.*;
import java.util.*;

public class Solution {

    static int[] manacher(char[] a) {
        int n = a.length;
        int[] p = new int[n];

        int center = 0;
        int right = 0;

        for (int i = 0; i < n; i++) {

            int mirror = 2 * center - i;

            if (i < right) {
                p[i] = Math.min(right - i, p[mirror]);
            }

            while (i - p[i] - 1 >= 0 &&
                   i + p[i] + 1 < n &&
                   a[i - p[i] - 1] == a[i + p[i] + 1]) {
                p[i]++;
            }

            if (i + p[i] > right) {
                center = i;
                right = i + p[i];
            }
        }

        return p;
    }

    static class SparseTable {
        int[][] table;
        int[] log;

        SparseTable(int[] arr) {
            int n = arr.length;

            log = new int[n + 1];

            for (int i = 2; i <= n; i++) {
                log[i] = log[i / 2] + 1;
            }

            int k = log[n] + 1;
            table = new int[k][n];

            System.arraycopy(arr, 0, table[0], 0, n);

            for (int j = 1; j < k; j++) {
                for (int i = 0; i + (1 << j) <= n; i++) {
                    table[j][i] = Math.max(
                        table[j - 1][i],
                        table[j - 1][i + (1 << (j - 1))]
                    );
                }
            }
        }

        int query(int left, int right) {
            if (left > right) {
                return 0;
            }

            int len = right - left + 1;
            int k = log[len];

            return Math.max(
                table[k][left],
                table[k][right - (1 << k) + 1]
            );
        }
    }

    public static void main(String[] args) throws Exception {

        BufferedReader br =
            new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine().trim());
        String s = br.readLine().trim();

        // Duplicate the string
        String doubled = s + s;

        /*
         * Transform:
         * abc -> #a#b#c#
         *
         * This allows us to handle both odd and even
         * length palindromes with one Manacher array.
         */
        int m = doubled.length() * 2 + 1;
        char[] transformed = new char[m];

        for (int i = 0; i < m; i++) {
            if (i % 2 == 0) {
                transformed[i] = '#';
            } else {
                transformed[i] = doubled.charAt(i / 2);
            }
        }

        int[] radius = manacher(transformed);

        /*
         * We only need palindrome centers that can occur
         * inside the doubled string.
         *
         * Limit every palindrome to length n because a
         * rotation contains only n characters.
         */
        for (int i = 0; i < radius.length; i++) {
            radius[i] = Math.min(radius[i], n);
        }

        SparseTable st = new SparseTable(radius);

        StringBuilder out = new StringBuilder();

        for (int start = 0; start < n; start++) {

            /*
             * In transformed string:
             *
             * character position x -> 2*x + 1
             *
             * The rotation occupies:
             * start ... start+n-1
             */
            int left = 2 * start + 1;
            int right = 2 * (start + n - 1) + 1;

            int low = 1;
            int high = n;
            int answer = 1;

            /*
             * Binary search the maximum palindrome length.
             */
            while (low <= high) {

                int len = (low + high) / 2;

                /*
                 * A palindrome of length len must have its
                 * center between these two boundaries.
                 */
                int centerLeft = left + len - 1;
                int centerRight = right - len + 1;

                if (centerLeft <= centerRight &&
                    st.query(centerLeft, centerRight) >= len) {

                    answer = len;
                    low = len + 1;

                } else {
                    high = len - 1;
                }
            }

            out.append(answer).append('\n');
        }

        System.out.print(out);
    }
}
